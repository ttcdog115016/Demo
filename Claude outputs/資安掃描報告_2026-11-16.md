- 指令（`SCR` 為暫存資料夾，輸出檔不放入 repo）：

```bash
  --exclude node_modules --exclude doc_site --exclude "Claude outputs" --exclude sbom --exclude slides \
  --json -o out/semgrep.json src packaging docker backend/app scripts e2e
bandit -r backend/app -f json -o out/bandit_be.json -q

pip-audit -r backend/requirements.txt --no-deps --disable-pip -f json -o out/pipaudit_req.json
gitleaks git . --redact -f json -r out/gitleaks_git.json
gitleaks dir . --redact -f json -r out/gitleaks_dir.json
osv-scanner scan source -L package-lock.json --format json --output-file out/osv_fe.json
osv-scanner scan source -L backend/uv.lock --format json --output-file out/osv_be.json
```
